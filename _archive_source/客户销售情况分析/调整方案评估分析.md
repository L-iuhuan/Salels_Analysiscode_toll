def calc_churn_warning_v2(...):
    # 原有逻辑
    基础预警 = 距上次采购天数 > 间隔 × 1.5
    
    # 新增：采购量豁免检查
    前两季度月均量 = 前9个月的总量 / 活跃月数
    本季度月均量 = (本月量 + 前两月量) / 活跃月数
    
    if 上季度最大月量 > 前两季度月均量 × 2:  # 季末大量采购
        采购量下降比 = 本季度月均量 / 前两季度月均量
        if 采购量下降比 >= 0.8:
            豁免触发  # 大量采购后正常回落
        elif 采购量下降比 < 0.2:
            减弱预警  # 改为提示级别而非触发级别
