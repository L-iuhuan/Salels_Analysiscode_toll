## 1. 数据异常诊断

### 1.1 输出数据异常清单

通过检查 `output/gold/客户全景.csv` (310 rows)、`output/silver/` (4 CSV)、`output/gold/异常日志.csv` 发现以下异常：

| # | 异常项 | 数值 | 严重度 | 原因分析 |
|---|--------|------|--------|----------|
| A1 | **流失期客户占比 44.5%** | 138/310 | 🔴 高 | 数据原因：大量客户确实已停采超12个月。代码未过滤这些客户参与评分（正确），但评分卡中使用了`SCORE_INACTIVE_DEFAULT=25`回退 |
| A2 | **客户等级"未知" 40.3%** | 125/310 | 🟡 中 | 数据原因：CRM匹配率~60%，未匹配客户标"未知"。代码中的 `CUSTOMER_TIER_MAP.fallback_thresholds` 已废弃为空列表 |
| A3 | **渠道类型：代理96% vs 直供4%** | 299 vs 11 | 🟡 中 | 数据原因+代码原因：`CHANNEL_DERIVE` 规则推导覆盖100%客户，但业务数据本身代理占绝对主导。直供仅11个客户，渠道隔离评分(`qcut`需要>=5)对直供渠道回退为统一评分 |
| A4 | **终端CRM字段 39%空值** | 20列均39% | 🟡 中 | 数据原因：CRM覆盖率~60%，这与方案文档描述的完全一致 |
| A5 | **采购中断预警 31.9%** | 99/310 True | 🟡 中 | 代码原因+数据原因：`calc_churn_warning()`使用月×30天近似，可能对2月/31天月份有偏差。部分客户因自然业务周期导致"假阳性" |
| A6 | **异常日志 445条** | 营收断崖171 + 采购中断155 | 🟡 中 | 代码原因：`detect_revenue_cliff` 阈值 `revenue_cliff_ratio=0.3` 可能过于敏感，产生过多中等级异常 |
| A7 | **Silver层80行负利润** | 80条 `profit_clip_sum<0` | 🟢 低 | 代码原因：`winsorize_margins` 已做-50%下限钳制，但允许真实亏损保留（这是设计意图）。需文档说明 |
| A8 | **收入增长率 极值30.85** | max=3085% | 🟢 低 | 数据原因：部分客户从极低基数暴涨。`portrait.py:_dim_momentum` 未对增长率做钳制（仅product_lifecycle有截断） |
| A9 | **近12月毛利为负** | 部分客户 | 🟢 低 | 数据原因+代码原因：`_dim_momentum` 中当 `profit_clip_sum` 为负时未特别处理，导致负毛利率直接进入评分卡 |
| A10 | **_CRM渠道类型 91%空值** | 282/310 NaN | 🟢 低 | 代码原因：`_CRM渠道类型` 是 `customer_master.py` 注入的临时列，仅在CRM有明确渠道映射时才有值，这是设计行为 |


### 3.2 抽象接口设计（Protocol类）

```python
# core/interfaces.py
from typing import Protocol, Dict, Any, Optional
import pandas as pd

# ─── 数据加载 ───
class IDataLoader(Protocol):
    """数据加载器接口。"""
    def load(self, path: str, **kwargs) -> pd.DataFrame: ...
    def find_source(self, data_dir: str) -> str: ...

# ─── 数据清洗 ───
class IDataCleaner(Protocol):
    """数据清洗器接口。"""
    def clean(self, df: pd.DataFrame) -> pd.DataFrame: ...
    def validate(self, df: pd.DataFrame) -> list[str]: ...  # 返回错误列表

# ─── 数据聚合 ───
class IDataAggregator(Protocol):
    """数据聚合器接口。"""
    def aggregate(self, df: pd.DataFrame) -> dict[str, pd.DataFrame]: ...

# ─── Silver层构建 ───
class ISilverBuilder(Protocol):
    """Silver层构建器接口。"""
    def build(self, source_path: str) -> dict[str, pd.DataFrame]: ...

# ─── 分析器 ───
class IAnalyzer(Protocol):
    """分析器接口。"""
    def analyze(self, data: dict[str, pd.DataFrame]) -> pd.DataFrame: ...

# ─── 评分器 ───
class IScorer(Protocol):
    """评分器接口。"""
    def score(self, portrait: pd.DataFrame) -> pd.DataFrame: ...

# ─── Gold层生成 ───
class IGoldGenerator(Protocol):
    """Gold层生成器接口。"""
    def generate(self, portrait: pd.DataFrame, silver: dict, **deps) -> dict[str, pd.DataFrame]: ...

# ─── 报告生成 ───
class IReporter(Protocol):
    """报告生成器接口。"""
    def generate(self, gold: dict[str, pd.DataFrame]) -> str: ...

# ─── 验证器 ───
class IValidator(Protocol):
    """数据验证器接口。"""
    def validate(self, df: pd.DataFrame, stage: str) -> dict[str, Any]: ...
```

### 3.3 Pipeline 依赖注入容器

```python
# core/pipeline.py
from dataclasses import dataclass, field
from typing import Optional, Type
from core.interfaces import (
    IDataLoader, IDataCleaner, IDataAggregator, ISilverBuilder,
    IAnalyzer, IScorer, IGoldGenerator, IReporter, IValidator,
)
from core.config import AppConfig


@dataclass
class Pipeline:
    """半导体分析管线 — 依赖注入容器。

    所有依赖通过构造函数注入，而不是在模块内部直接 import。
    每个阶段的处理函数接收其依赖作为参数。

    Usage:
        config = AppConfig.from_yaml("config/settings.yaml")
        pipeline = Pipeline(config=config)
        pipeline.run(stages=["silver", "product", "customer"])
    """
    config: AppConfig

    # 可注入的依赖（默认使用生产实现）
    data_loader: IDataLoader = None
    data_cleaner: IDataCleaner = None
    data_aggregator: IDataAggregator = None
    silver_builder: ISilverBuilder = None
    product_analyzer: IAnalyzer = None
    customer_analyzer: IAnalyzer = None
    scorer: IScorer = None
    gold_generator: IGoldGenerator = None
    reporter: IReporter = None
    validator: IValidator = None
    timing_enabled: bool = False

    def __post_init__(self):
        """自动装配默认实现。"""
        from data_pipeline.loader import ExcelDataLoader
        from data_pipeline.cleaner import DefaultCleaner
        from data_pipeline.aggregator import DefaultAggregator
        # ... etc

        self.data_loader = self.data_loader or ExcelDataLoader()
        self.data_cleaner = self.data_cleaner or DefaultCleaner(self.config)
        self.data_aggregator = self.data_aggregator or DefaultAggregator()
        # ...

    def run(self, stages: list[str] = None, source_path: str = None):
        """执行管线。"""
        stages = stages or self.config.run_stages

        # Stage 1: Silver
        raw = self.data_loader.load(source_path or self._find_source())
        validation_errors = self.validator.validate(raw, "raw")
        if validation_errors:
            self._log_validation(validation_errors)

        cleaned = self.data_cleaner.clean(raw)
        silver = self.data_aggregator.aggregate(cleaned)

        # Stage 2a: Product
        product_result = self.product_analyzer.analyze(silver)
        # ... etc

    def _find_source(self) -> str:
        return self.data_loader.find_source(self.config.data_dir)

    def _log_validation(self, errors: dict):
        """记录验证错误到日志。"""
        pass
```

```python
    profit_clip_sum: Series[float] = pa.Field()
    order_count: Series[int] = pa.Field(ge=0)

    @pa.dataframe_check
    def margin_range(cls, df):
        """毛利率应在合理范围。"""
        mask = (df["rev_sum"] > 0) & df["profit_clip_sum"].notna()
        if mask.sum() == 0:
            return True
        margins = df.loc[mask, "profit_clip_sum"] / df.loc[mask, "rev_sum"] * 100
```

### 5.3 验证器实现

```python
# data_pipeline/validator.py
import pandera as pa
from pandera.errors import SchemaErrors
from typing import Dict, Any
from config.schema import RawSchema, CleanSchema, SilverCustomerMonthlySchema, GoldCustomerProfileSchema


class DataValidator:
    """数据验证器 — 在管线的每个阶段插入验证节点。"""

    SCHEMAS = {
        "raw": RawSchema,
        "clean": CleanSchema,
        "silver_customer_monthly": SilverCustomerMonthlySchema,
        "gold_customer_profile": GoldCustomerProfileSchema,
    }

    def validate(self, df, stage: str, strict: bool = False) -> Dict[str, Any]:
        """验证DataFrame是否符合指定阶段的Schema。

        Returns:
            dict: {"valid": bool, "errors": list[str], "warnings": list[str], "stats": dict}
        """
        schema = self.SCHEMAS.get(stage)
        if schema is None:
            return {"valid": True, "errors": [], "warnings": [f"Unknown stage: {stage}"], "stats": {}}

        result = {"valid": True, "errors": [], "warnings": [], "stats": {}}

        try:
            schema.validate(df, lazy=True)
        except SchemaErrors as e:
            result["valid"] = False
            result["errors"] = [
                f"{err.schema_context}: {err.reason_code} - column={err.column}"
                for err in e.schema_errors
            ]
            if strict:
                raise

        # 补充统计信息
        result["stats"] = {
            "row_count": len(df),
            "col_count": len(df.columns),
            "null_pct": round(df.isnull().mean().mean() * 100, 1),
        }

        return result

    def validate_all(self, data: Dict[str, Any]) -> Dict[str, Dict]:
        """批量验证所有阶段数据。"""
        stages = {
            k: v for k, v in data.items()
            if isinstance(v, pd.DataFrame) and k in self.SCHEMAS
        }
        return {stage: self.validate(df, stage) for stage, df in stages.items()}
```

### 5.4 纯Pandas验证（无Pandera依赖）

若不想引入 Pandera 依赖，可用纯 pandas 实现：

```python
# data_pipeline/validator_simple.py
import pandas as pd
import numpy as np
from typing import List, Dict, Any, Tuple


class SimpleValidator:
    """轻量级验证器（纯pandas，无额外依赖）。"""

    @staticmethod
    def check_required_columns(df: pd.DataFrame, required: List[str]) -> List[str]:
        """检查必填列是否存在。"""
        return [c for c in required if c not in df.columns]

    @staticmethod
    def check_null_constraint(df: pd.DataFrame, cols: List[str], max_null_pct: float = 0.50) -> List[str]:
        """检查指定列的空值率是否超过阈值。"""
        issues = []
        for col in cols:
            if col in df.columns:
                null_pct = df[col].isna().mean()
                if null_pct > max_null_pct:
                    issues.append(f"{col}: {null_pct:.1%} null (threshold: {max_null_pct:.1%})")
        return issues

    @staticmethod
    def check_value_range(df: pd.DataFrame, col: str, lo: float = None, hi: float = None) -> List[str]:
        """检查数值列的值范围。"""
        if col not in df.columns:
            return [f"{col}: column missing"]
        s = df[col].dropna()
        issues = []
        if lo is not None and (s < lo).any():
            n = (s < lo).sum()
            issues.append(f"{col}: {n} rows below {lo} (min={s.min():.2f})")
        if hi is not None and (s > hi).any():
            n = (s > hi).sum()
            issues.append(f"{col}: {n} rows above {hi} (max={s.max():.2f})")
        return issues

    @staticmethod
    def check_category_values(df: pd.DataFrame, col: str, allowed: List[str]) -> List[str]:
        """检查分类列的允许值。"""
        if col not in df.columns:
            return [f"{col}: column missing"]
        actual = set(df[col].dropna().unique())
        invalid = actual - set(allowed)
        if invalid:
            return [f"{col}: invalid values {invalid}"]
        return []

    def validate_raw(self, df: pd.DataFrame) -> Dict[str, Any]:
        """验证源数据。"""
        issues = []
        issues += self.check_required_columns(df, ["客户编号", "产品品种", "数量", "金额", "发货日期"])
        issues += self.check_null_constraint(df, ["客户编号", "发货日期"])
        issues += self.check_value_range(df, "金额", lo=0)
        return {"valid": len(issues) == 0, "issues": issues}

    def validate_silver(self, silver: Dict[str, pd.DataFrame]) -> Dict[str, Any]:
        """验证Silver层。"""
        results = {}
        for key, df in silver.items():
            issues = []
            issues += self.check_required_columns(df, ["_月", "rev_sum", "qty_sum"])
            issues += self.check_value_range(df, "rev_sum", lo=0)
            issues += self.check_value_range(df, "qty_sum", lo=0)
            results[key] = {"valid": len(issues) == 0, "issues": issues}
        return results
```

```python
    # 3. 行数过低检查
    if len(df) < 10:
        issues.append(f"数据行数过少 ({len(df)} 行)，可能数据不完整")

    # 4. 日期范围检查
    date_cols = [c for c in df.columns if "日期" in str(c)]
    for dc in date_cols:
        try:
            dates = pd.to_datetime(df[dc], errors='coerce')
            na_rate = dates.isna().mean()
            if na_rate > 0.5:
                issues.append(f"{dc}: {na_rate:.0%} 日期解析失败")
        except Exception:
            issues.append(f"{dc}: 日期列无法解析")

    return len(issues) == 0, issues
```
